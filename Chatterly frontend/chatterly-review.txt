import { createContext, useCallback, useContext, useEffect, useMemo, useState } from "react";

const ThemeContext = createContext(null);
const STORAGE_KEY = "cloudchat-theme";

function getInitialTheme() {
  try {
    const saved = localStorage.getItem(STORAGE_KEY);
    if (saved === "light" || saved === "dark") return saved;
  } catch {
    /* storage may be blocked; fall through to system preference */
  }
  return window.matchMedia?.("(prefers-color-scheme: dark)").matches ? "dark" : "light";
}

export function ThemeProvider({ children }) {
  const [theme, setTheme] = useState(getInitialTheme);

  useEffect(() => {
    document.documentElement.setAttribute("data-theme", theme);
    try {
      localStorage.setItem(STORAGE_KEY, theme); // UI preference only, never credentials
    } catch {
      /* ignore */
    }
  }, [theme]);

  const toggleTheme = useCallback(() => setTheme((t) => (t === "light" ? "dark" : "light")), []);
  const value = useMemo(() => ({ theme, setTheme, toggleTheme }), [theme, toggleTheme]);

  return <ThemeContext.Provider value={value}>{children}</ThemeContext.Provider>;
}

export function useTheme() {
  const ctx = useContext(ThemeContext);
  if (!ctx) throw new Error("useTheme must be used inside <ThemeProvider>");
  return ctx;
}

import { createContext, useCallback, useContext, useEffect, useMemo, useState } from "react";
import * as authService from "../services/authService.js";

/*
 * Auth state lives in memory only. Passwords and tokens are never written to
 * localStorage. With the future Django backend, the session should travel in an
 * HttpOnly cookie that the browser manages, so this file never touches a token.
 * Today the sign-in is a simulation: no real account or session exists.
 */
const AuthContext = createContext(null);

export function AuthProvider({ children }) {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    let cancelled = false;
    authService
      .getCurrentUser()
      .then((u) => !cancelled && setUser(u))
      .catch(() => !cancelled && setUser(null))
      .finally(() => !cancelled && setLoading(false));
    return () => { cancelled = true; };
  }, []);

  const signIn = useCallback(async (credentials) => {
    const u = await authService.signIn(credentials);
    setUser(u);
    return u;
  }, []);

  const signUp = useCallback(async (details) => {
    const u = await authService.signUp(details);
    setUser(u);
    return u;
  }, []);

  const signOut = useCallback(async () => {
    await authService.signOut();
    setUser(null);
  }, []);

  const value = useMemo(
    () => ({ user, setUser, loading, isAuthenticated: Boolean(user), signIn, signUp, signOut }),
    [user, loading, signIn, signUp, signOut]
  );

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
}

export function useAuth() {
  const ctx = useContext(AuthContext);
  if (!ctx) throw new Error("useAuth must be used inside <AuthProvider>");
  return ctx;
}

import { createContext, useCallback, useContext, useMemo, useRef, useState } from "react";
import ToastRegion from "../components/common/Toast.jsx";

const ToastContext = createContext(null);

export function ToastProvider({ children }) {
  const [toasts, setToasts] = useState([]);
  const nextId = useRef(1);

  const dismissToast = useCallback((id) => {
    setToasts((list) => list.filter((t) => t.id !== id));
  }, []);

  const showToast = useCallback(
    (message, type = "info", duration = 4000) => {
      const id = nextId.current++;
      setToasts((list) => [...list, { id, message, type }]);
      if (duration > 0) setTimeout(() => dismissToast(id), duration);
      return id;
    },
    [dismissToast]
  );

  const value = useMemo(() => ({ showToast, dismissToast }), [showToast, dismissToast]);

  return (
    <ToastContext.Provider value={value}>
      {children}
      <ToastRegion toasts={toasts} onDismiss={dismissToast} />
    </ToastContext.Provider>
  );
}

export function useToast() {
  const ctx = useContext(ToastContext);
  if (!ctx) throw new Error("useToast must be used inside <ToastProvider>");
  return ctx;
}

import { useCallback, useEffect, useState } from "react";
import { getConversations } from "../services/chatService.js";

/** Loads the conversation list with loading and error state. */
export function useChats() {
  const [chats, setChats] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  const load = useCallback(async () => {
    setLoading(true);
    setError(null);
    try {
      setChats(await getConversations());
    } catch (err) {
      setError(err.message);
    } finally {
      setLoading(false);
    }
  }, []);

  useEffect(() => { load(); }, [load]);

  return { chats, setChats, loading, error, reload: load };
}

import { useCallback, useEffect, useRef, useState } from "react";
import { getMessages } from "../services/messageService.js";

/** Loads messages for one conversation; reloads when the conversation changes. */
export function useMessages(conversationId) {
  const [messages, setMessages] = useState([]);
  const [loading, setLoading] = useState(Boolean(conversationId));
  const [error, setError] = useState(null);
  const latest = useRef(0);

  const load = useCallback(async () => {
    const ticket = ++latest.current; // ignore responses from conversations we already left
    if (!conversationId) { setMessages([]); setLoading(false); return; }
    setLoading(true);
    setError(null);
    try {
      const data = await getMessages(conversationId);
      if (ticket === latest.current) setMessages(data);
    } catch (err) {
      if (ticket === latest.current) setError(err.message);
    } finally {
      if (ticket === latest.current) setLoading(false);
    }
  }, [conversationId]);

  useEffect(() => { load(); }, [load]);

  return { messages, setMessages, loading, error, reload: load };
}

const COLORS = ["#0e7490", "#047857", "#4d7c0f", "#b45309", "#be185d", "#6d28d9", "#1d4ed8", "#b91c1c"];

function hash(text) {
  let h = 0;
  for (let i = 0; i < text.length; i++) h = (h * 31 + text.charCodeAt(i)) >>> 0;
  return h;
}

function initials(name) {
  const parts = name.trim().split(/\s+/).filter(Boolean);
  if (parts.length === 0) return "?";
  const first = parts[0][0];
  const last = parts.length > 1 ? parts[parts.length - 1][0] : "";
  return (first + last).toUpperCase();
}

/** Round avatar with a deterministic colour and optional online dot. */
export default function Avatar({ name = "", src, size = "md", online }) {
  const style = { background: COLORS[hash(name) % COLORS.length] };
  return (
    <span className={`avatar avatar-${size}`} style={src ? undefined : style} role="img" aria-label={name}>
      {src ? <img src={src} alt="" /> : initials(name)}
      {online !== undefined && (
        <span
          className="avatar-status"
          style={online ? undefined : { background: "var(--text-muted)" }}
          title={online ? "Online" : "Offline"}
        />
      )}
    </span>
  );
}

import { useEffect, useId, useRef } from "react";
import { createPortal } from "react-dom";
import { X } from "lucide-react";

/** Accessible dialog: closes on Escape or backdrop click, locks page scroll, restores focus. */
export default function Modal({ open, title, onClose, children, footer }) {
  const titleId = useId();
  const dialogRef = useRef(null);

  useEffect(() => {
    if (!open) return;
    const previouslyFocused = document.activeElement;
    const previousOverflow = document.body.style.overflow;
    document.body.style.overflow = "hidden";
    dialogRef.current?.focus();

    const onKeyDown = (e) => e.key === "Escape" && onClose();
    document.addEventListener("keydown", onKeyDown);

    return () => {
      document.removeEventListener("keydown", onKeyDown);
      document.body.style.overflow = previousOverflow;
      previouslyFocused?.focus?.();
    };
  }, [open, onClose]);

  if (!open) return null;

  return createPortal(
    <div className="modal-backdrop" onMouseDown={(e) => e.target === e.currentTarget && onClose()}>
      <div className="modal" role="dialog" aria-modal="true" aria-labelledby={titleId} tabIndex={-1} ref={dialogRef}>
        <div className="modal-head">
          <h3 id={titleId}>{title}</h3>
          <button className="icon-btn" onClick={onClose} aria-label="Close dialog">
            <X size={20} />
          </button>
        </div>
        <div className="modal-body">{children}</div>
        {footer && <div className="modal-foot">{footer}</div>}
      </div>
    </div>,
    document.body
  );
}

/*
 * Single place that decides how the app talks to data.
 *
 * VITE_USE_MOCK defaults to true, so every service answers from src/data/mockData.js.
 * When the Django backend exists, set VITE_USE_MOCK=false and implement the
 * non-mock branch in each service. No real endpoint exists yet; the paths
 * noted in the services are PLANNED, not available.
 */
export const USE_MOCK = import.meta.env.VITE_USE_MOCK !== "false";
export const API_BASE_URL = import.meta.env.VITE_API_BASE_URL ?? "";
export const WS_URL = import.meta.env.VITE_WS_URL ?? "";

export class ApiError extends Error {
  constructor(message, status = 0) {
    super(message);
    this.name = "ApiError";
    this.status = status;
  }
}

/** Simulated network latency for mock mode. */
export const delay = (ms = 450) => new Promise((resolve) => setTimeout(resolve, ms));

/** Thrown by the non-mock branches until the backend is built. */
export function notConnected(operation) {
  return new ApiError(`${operation} is not connected to a backend yet.`, 501);
}

/**
 * Thin fetch wrapper for the future backend. Uses cookies for the session
 * (credentials: "include") so no token is ever stored in JavaScript.
 */
export async function request(path, { method = "GET", body, headers = {} } = {}) {
  let response;
  try {
    response = await fetch(`${API_BASE_URL}${path}`, {
      method,
      credentials: "include",
      headers: { "Content-Type": "application/json", ...headers },
      body: body ? JSON.stringify(body) : undefined,
    });
  } catch {
    throw new ApiError("Could not reach the server. Check your connection and try again.");
  }
  if (!response.ok) throw new ApiError("The request failed.", response.status);
  return response.status === 204 ? null : response.json();
}

import { USE_MOCK, WS_URL } from "./api.js";

/*
 * Real-time integration point for Django Channels.
 * In mock mode this returns a no-op object with the same shape, so components
 * can subscribe today and keep working once a real connection replaces it.
 * Nothing is delivered in real time until the backend exists.
 */
export function createChatSocket({ onMessage, onStatusChange } = {}) {
  if (USE_MOCK || !WS_URL) {
    return { connect() { onStatusChange?.("mock"); }, disconnect() {}, send() {} };
  }

  let ws;
  return {
    connect() {
      ws = new WebSocket(WS_URL); // PLANNED: final path and auth handshake decided with the backend
      ws.onopen = () => onStatusChange?.("open");
      ws.onclose = () => onStatusChange?.("closed");
      ws.onmessage = (event) => {
        try { onMessage?.(JSON.parse(event.data)); } catch { /* ignore malformed frames */ }
      };
    },
    disconnect() { ws?.close(); },
    send(payload) { if (ws?.readyState === WebSocket.OPEN) ws.send(JSON.stringify(payload)); },
  };
}
